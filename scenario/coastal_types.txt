processor CT_HUNTER_GUIDE WSF_SCRIPT_PROCESSOR
   update_interval 0.25 sec
   on_initialize
      WsfPlatform left = WsfSimulation.FindPlatform("hunter_1");
      WsfPlatform right = WsfSimulation.FindPlatform("hunter_2");
      double leftRange = 1000000;
      double rightRange = 1000000;
      if (left.IsValid()) leftRange = PLATFORM.GroundRangeTo(left);
      if (right.IsValid()) rightRange = PLATFORM.GroundRangeTo(right);
      if (leftRange <= rightRange)
      {
         PLATFORM.SetAuxData("cue_lat", gH1Lat);
         PLATFORM.SetAuxData("cue_lon", gH1Lon);
         PLATFORM.SetAuxData("cue_alt", gH1Alt);
         PLATFORM.SetAuxData("owner", gH1Owner);
         PLATFORM.SetAuxData("track_name", gH1Track);
         PLATFORM.SetAuxData("num", gH1Num);
      }
      else
      {
         PLATFORM.SetAuxData("cue_lat", gH2Lat);
         PLATFORM.SetAuxData("cue_lon", gH2Lon);
         PLATFORM.SetAuxData("cue_alt", gH2Alt);
         PLATFORM.SetAuxData("owner", gH2Owner);
         PLATFORM.SetAuxData("track_name", gH2Track);
         PLATFORM.SetAuxData("num", gH2Num);
      }
   end_on_initialize
   on_update
      if (PLATFORM.TimeSinceCreation() > 120)
      {
         PLATFORM.SetDamageFactor(1);
         return;
      }
      bool terminal = false;
      double lat = 0;
      double lon = 0;
      double alt = 0;
      foreach (WsfTrack track in PLATFORM.MasterRawTrackList())
      {
         if (terminal || !track.IsValid() || !track.LocationValid()) continue;
         if (track.SensorName() != "seeker") continue;
         lat = track.Latitude();
         lon = track.Longitude();
         alt = track.Altitude();
         terminal = true;
      }
      if (!terminal && PLATFORM.AuxDataExists("owner"))
      {
         WsfPlatform owner = WsfSimulation.FindPlatform(PLATFORM.AuxDataString("owner"));
         double want = PLATFORM.AuxDataDouble("num");
         string wantName = PLATFORM.AuxDataString("track_name");
         if (owner.IsValid())
         {
            foreach (WsfTrack track in owner.MasterRawTrackList())
            {
               if (terminal || !track.IsValid() || !track.LocationValid()) continue;
               double idNumber = (double) track.TrackId().Number();
               if (track.TrackId().Name() != wantName) continue;
               if (idNumber < want - 0.5 || idNumber > want + 0.5) continue;
               lat = track.Latitude();
               lon = track.Longitude();
               alt = track.Altitude();
               terminal = true;
            }
         }
      }
      if (!terminal && PLATFORM.AuxDataExists("cue_lat"))
      {
         lat = PLATFORM.AuxDataDouble("cue_lat");
         lon = PLATFORM.AuxDataDouble("cue_lon");
         alt = PLATFORM.AuxDataDouble("cue_alt");
         if (lat != 0 || lon != 0) terminal = true;
      }
      if (terminal)
      {
         PLATFORM.GoToSpeed(60);
         PLATFORM.GoToLocation(lat, lon, alt);
      }
   end_on_update
end_processor

processor CT_UAV_BRAIN WSF_SCRIPT_PROCESSOR
   update_interval 0.5 sec
   on_update
      if (PLATFORM.DamageFactor() >= 1) return;
      string me = PLATFORM.Name();
      bool held = false;
      WsfPlatform jammer = WsfSimulation.FindPlatform("jam_1");
      if (gJam1 == me && jammer.IsValid() && jammer.GroundRangeTo(PLATFORM) <= 1000)
      {
         held = true;
      }
      jammer = WsfSimulation.FindPlatform("jam_2");
      if (gJam2 == me && jammer.IsValid() && jammer.GroundRangeTo(PLATFORM) <= 1000)
      {
         held = true;
      }
      if (held)
      {
         if (!PLATFORM.AuxDataExists("jam_mark") || PLATFORM.AuxDataDouble("jam_mark") < 1)
         {
            PLATFORM.SetAuxData("jam_mark", 1.0);
            writeln("CT_JAM_HOLD ", me, " ", TIME_NOW);
         }
         PLATFORM.SetAuxData("jam_t", TIME_NOW);
         PLATFORM.GoToSpeed(5);
         PLATFORM.GoToLocation(PLATFORM.Latitude(), PLATFORM.Longitude(), PLATFORM.Altitude());
      }
      else
      {
         if (PLATFORM.AuxDataExists("jam_t") && PLATFORM.AuxDataDouble("jam_mark") > 0)
         {
            if (TIME_NOW < PLATFORM.AuxDataDouble("jam_t") + 5)
            {
               PLATFORM.GoToSpeed(5);
               PLATFORM.GoToLocation(PLATFORM.Latitude(), PLATFORM.Longitude(), PLATFORM.Altitude());
            }
            else
            {
               PLATFORM.SetAuxData("jam_mark", 0.0);
               writeln("CT_JAM_RELEASE ", me, " ", TIME_NOW);
               PLATFORM.GoToSpeed(40);
               PLATFORM.GoToLocation(PLATFORM.AuxDataDouble("dest_lat"), PLATFORM.AuxDataDouble("dest_lon"), PLATFORM.AuxDataDouble("dest_alt"));
            }
         }
      }
      for (int i = 0; i < WsfSimulation.PlatformCount(); i = i + 1)
      {
         WsfPlatform site = WsfSimulation.PlatformEntry(i);
         if (!site.IsValid() || site.Side() != "gold") continue;
         if (site.DamageFactor() >= 1) continue;
         if (PLATFORM.GroundRangeTo(site) > 40) continue;
         if (PLATFORM.Altitude() - site.Altitude() > 80) continue;
         writeln("CT_IMPACT ", me, " ", site.Name(), " ", TIME_NOW);
         site.SetDamageFactor(1);
         PLATFORM.SetDamageFactor(1);
         return;
      }
   end_on_update
end_processor

processor CT_EO_CUE WSF_SCRIPT_PROCESSOR
   update_interval 0.5 sec
   on_update
      if (!PLATFORM.AuxDataExists("cue_num")) return;
      WsfPlatform owner = WsfSimulation.FindPlatform(PLATFORM.AuxDataString("cue_owner"));
      if (!owner.IsValid()) return;
      string wantName = PLATFORM.AuxDataString("cue_track");
      double want = PLATFORM.AuxDataDouble("cue_num");
      foreach (WsfTrack track in owner.MasterRawTrackList())
      {
         if (!track.IsValid() || !track.LocationValid()) continue;
         if (track.TrackId().Name() != wantName) continue;
         double idNumber = (double) track.TrackId().Number();
         if (idNumber < want - 0.5 || idNumber > want + 0.5) continue;
         if (PLATFORM.GroundRangeTo(track) > 3000) continue;
         double rel = PLATFORM.TrueBearingTo(track) - PLATFORM.Heading();
         if (rel > 180) rel = rel - 360;
         if (rel < -180) rel = rel + 360;
         WsfSensor eye = PLATFORM.Sensor("eo");
         if (eye.IsValid()) eye.SetYaw(rel);
      }
   end_on_update
end_processor
